    })
  })

  })

  });
};

// countSeedlings — supports BOTH the new flat `seedlings` array shape and
// the legacy fruitSeedlings/forestSeedlings/medicinalSeedlings shape.
function countSeedlings(s){
  var r={fruit:0,forest:0,medicinal:0};
  if(s && Array.isArray(s.seedlings)){
    s.seedlings.forEach(function(e){
      var qty=parseInt(e.quantity)||0;
      var cat=(e.category||"").trim();
      if(cat.indexOf("ফল")===0||cat==="fruit") r.fruit+=qty;
      else if(cat.indexOf("বন")===0||cat==="forest") r.forest+=qty;
      else if(cat.indexOf("ঔষ")===0||cat==="medicinal") r.medicinal+=qty;
      else r.fruit+=qty;
    });
    return r;
  }
  ["fruitSeedlings","forestSeedlings","medicinalSeedlings"].forEach(function(k,i){
    (s[k]||[]).forEach(function(e){
      r[["fruit","forest","medicinal"][i]]+=(parseInt(e.count)||0)+(parseInt(e.graftingCount)||0);
    });
  });
  return r;
}

// Returns ALL entries that belong to the logged-in user — merging the device's
// local submissions with the national App_Entry sheet, de-duplicated by
// submissionId, and matching on ANY of the three mobile fields (farmer, SAAO,
// or monitoring officer). Previously this only matched farmerMobile against
// local submissions, which silently hid entries the user had synced from
// another device or had recorded as the monitoring officer rather than the
// planter — those now correctly appear in the Profile tab too.
function getMySubmissionsBase(){
  var p=getProfile();
  var mob=p&&p.mobile?p.mobile:"";
  if(!mob){var el=document.getElementById("lookupMobile");mob=el?el.value.trim():""}
  if(!mob) return [];
  var matchesMob=function(s){
    return (s.farmerMobile||s.mobile)===mob ||
           s.saaoMobile===mob ||
           s.officerMobile===mob ||
           s.caretakerMobile===mob;
  };
  var byId={},merged=[];
  // National (server-synced) entries first — these are the authoritative copies
  // for anything that has already been pushed to the App_Entry sheet.
  (_nationalEntries||[]).forEach(function(s){
    if(!matchesMob(s)) return;
    var key=s.submissionId||s.id;
    if(key) byId[key]=true;
    s._source=s._source||"national";
    merged.push(s);
  });
  // Then local-only entries that haven't been synced yet (or whose sync failed).
  getSubmissions().forEach(function(s){
    if(!matchesMob(s)) return;
    var key=s.submissionId||s.id;
    if(key&&byId[key]) return; // already included via the national copy
    s._source=s._source||"local";
    merged.push(s);
  });
  return merged;
}

function applyMyFilterSet(subs){
  var from=document.getElementById("myFilterFrom")?document.getElementById("myFilterFrom").value:"";
  var to=document.getElementById("myFilterTo")?document.getElementById("myFilterTo").value:"";
  var src=document.getElementById("myFilterSource")?document.getElementById("myFilterSource").value:"";
  var dist=document.getElementById("myFilterDistrict")?document.getElementById("myFilterDistrict").value:"";
  var upz=document.getElementById("myFilterUpazila")?document.getElementById("myFilterUpazila").value:"";
  return subs.filter(function(s){
    var d=s.plantingDate||(s.submittedAt?s.submittedAt.slice(0,10):"");
    if(from&&d&&d<from) return false;
    if(to&&d&&d>to) return false;
    if(src&&s.sourceType!==src) return false;
    if(dist&&s.district!==dist) return false;
    if(upz&&s.upazila!==upz) return false;
    return true;
  });
}

// Simple, clearly-labeled averages — NOT precise agronomic modeling.
var MY_SURVIVAL_RATE=0.80;
var MY_CO2_KG_PER_TREE_YEAR=20;
function computeProbableOutput(subs){
  var c={fruit:0,forest:0,medicinal:0};
  subs.forEach(function(s){var t=countSeedlings(s);c.fruit+=t.fruit;c.forest+=t.forest;c.medicinal+=t.medicinal});
  var totalTrees=c.fruit+c.forest+c.medicinal;
  var survivalEst=Math.round(totalTrees*MY_SURVIVAL_RATE);
  var co2Est=Math.round(survivalEst*MY_CO2_KG_PER_TREE_YEAR);
  var fruitBearingEst=Math.round(c.fruit*MY_SURVIVAL_RATE);
  return {totalTrees:totalTrees,survivalEst:survivalEst,co2Est:co2Est,fruitBearingEst:fruitBearingEst,cat:c};
}

function getDashEntries(){
  var r=document.getElementById("dashRegion")&&document.getElementById("dashRegion").value;
  var d=document.getElementById("dashDistrict")&&document.getElementById("dashDistrict").value;
  var u=document.getElementById("dashUpazila")&&document.getElementById("dashUpazila").value;
  var st=document.getElementById("dashSourceType")&&document.getElementById("dashSourceType").value;
  var from=document.getElementById("dashDateFrom")&&document.getElementById("dashDateFrom").value;
  var to=document.getElementById("dashDateTo")&&document.getElementById("dashDateTo").value;
  var byId={},merged=[];
  (_nationalEntries||[]).forEach(function(s){var key=s.submissionId||s.id;if(key)byId[key]=true;merged.push(s)});
  getSubmissions().forEach(function(s){var key=s.submissionId||s.id;if(key&&byId[key])return;s._source=s._source||"local";merged.push(s)});
  return merged.filter(function(s){
    if(r&&s.region!==r)return false;
    if(d&&s.district!==d)return false;
    if(u&&s.upazila!==u)return false;
    if(st){
      // "অন্যান্য" (other) is a free-text bucket on the form — anything that
      // isn't one of the three fixed options counts as "other" here.
      if(st==="অন্যান্য"){ if(["প্রদর্শনী","প্রণোদনা","নিজস্ব"].indexOf(s.sourceType)!==-1)return false; }
      else if(s.sourceType!==st)return false;
    }
    if(from||to){
      var pD=s.plantingDate||(s.submittedAt?s.submittedAt.slice(0,10):"");
      if(from&&pD&&pD<from)return false;
      if(to&&pD&&pD>to)return false;
    }
    return true;
  });
}
