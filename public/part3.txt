// ── Google Sign-In (optional email auto-fetch) ──
// No-ops entirely until GOOGLE_CLIENT_ID above is replaced with a real one.
function _googleAuthConfigured(){ return GOOGLE_CLIENT_ID.indexOf("YOUR_")!==0; }
function _decodeGoogleJwt(token){
  try{
    var payload=token.split(".")[1];
    payload=payload.replace(/-/g,"+").replace(/_/g,"/");
    while(payload.length%4)payload+="=";
    return JSON.parse(decodeURIComponent(escape(atob(payload))));
  }catch(e){ console.error("Google JWT decode error:",e); return null; }
}
function initGoogleAuth(){
  if(!_googleAuthConfigured() || typeof google==="undefined" || !google.accounts || !google.accounts.id) return;
  try{
    google.accounts.id.initialize({ client_id: GOOGLE_CLIENT_ID, callback: handleGoogleCredential, auto_select: false });
    if(!document.getElementById("profileModal").classList.contains("hidden")) renderGoogleSignInButton();
  }catch(e){ console.error("Google auth init error:",e); }
}
function renderGoogleSignInButton(){
  if(!_googleAuthConfigured() || typeof google==="undefined" || !google.accounts || !google.accounts.id) return;
  var box=document.getElementById("googleSignInBox");
  if(!box) return;
  box.innerHTML="";
  try{
    google.accounts.id.renderButton(box,{ theme:"outline", size:"medium", text:"continue_with", width:260 });
  }catch(e){ console.error("Google button render error:",e); }
}
function handleGoogleCredential(response){
  var payload=_decodeGoogleJwt(response.credential);
  if(!payload || !payload.email) return;
  _pendingVerifiedEmail=payload.email;
  var emailField=document.getElementById("pEmail");
  if(emailField && !document.getElementById("profileModal").classList.contains("hidden")){
    emailField.value=payload.email;
    document.getElementById("pEmailVerifiedNote").classList.remove("hidden");
    if(payload.name && !_val("pName")) document.getElementById("pName").value=payload.name;
  } else {
    // Fired outside the profile modal -- this is the passive guest-visitor
    // path: log the ping so we can count unique visitors without requiring
    // guests to fill in the profile form.
    logVisitorPing(payload.email, isGuestMode()?"guest":"unprompted");
  }
}
// One Tap, once per browser session, so we can identify guest visitors
// (who by definition never open/complete the profile form) for analytics.
function attemptSilentVisitorPing(){
  if(!_googleAuthConfigured() || typeof google==="undefined" || !google.accounts || !google.accounts.id) return;
  if(sessionStorage.getItem(VISITOR_PING_KEY)==="1") return;
  sessionStorage.setItem(VISITOR_PING_KEY,"1");
  try{
    google.accounts.id.initialize({ client_id: GOOGLE_CLIENT_ID, callback: handleGoogleCredential, auto_select: true });
    google.accounts.id.prompt();
  }catch(e){ console.error("Silent visitor ping error:",e); }
}
function logVisitorPing(email, mode){
  try{
    fetch(GAS_SYNC_ENDPOINT,{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({
      entryType:"visitor_ping", email:email, mode:mode||"unknown", deviceId:getDeviceId()
    })}).catch(function(e){console.error("Visitor ping failed:",e)});
  }catch(e){console.error("Visitor ping error:",e)}
}

// The profile modal used to auto-open the instant this form loaded if no
// profile was saved yet -- with zero coordination with the React shell's
// WelcomeModal (the app's first-run intro screen, sometimes referred to as
// the "user guide"). That meant a first-time user could see the profile
// modal appear to open at the same moment as -- or immediately snap into
// view the instant they dismissed -- the welcome screen, with no visual
// hand-off between the two, which read as "two things opening at once."
// This now waits for an explicit "shell-ready" signal from the parent
// before opening, so the two are always strictly sequential: welcome
// screen first, profile modal only after. Loaded standalone (no parent
// shell, e.g. direct /legacy/plantation.html access) has nothing to wait
// for, so it keeps the original immediate behavior.
function maybeOpenProfileModalAfterShell(){
  if(window.self === window.top){
    openProfileModal(false);
    return;
  }
  var opened = false;
  var onShellReady = function(e){
    if(e.data && e.data.type === "shell-ready") openOnce();
  };
  var openOnce = function(){
    if(opened) return;
    opened = true;
    window.removeEventListener("message", onShellReady);
    // Re-check on arrival: time has passed since this was queued, and a
    // profile could now exist (or guest mode could now be active) from
    // something the user did while the welcome screen was still up.
    if(!getProfile() && !isGuestMode()) openProfileModal(false);
  };
  window.addEventListener("message", onShellReady);
  try{ window.parent.postMessage({ type:"iframe-ready" }, "*"); }catch(_){}
  // Safety net: an older/cached parent bundle (from before this fix) or
  // any other reason "shell-ready" never arrives should never permanently
  // block the profile modal from opening at all.
  setTimeout(openOnce, 2500);
}

// ── GPS → Division/Region/District/Upazila auto-select ──
// Nominatim already returns structured admin names in its response; the
// old code discarded everything except a flattened address string. This
// matches those names against our own BD/BD_UPAZILA/BD_DIVISIONS lists so
// the four cascading dropdowns fill themselves instead of the SAAO tapping
// through each one by hand.
function _norm(s){
  // .normalize('NFC') matters here: Nominatim's Bengali text and this
  // file's own BD_UPAZILA constant can encode visually-identical
  // characters (e.g. ড়) in different Unicode forms (precomposed vs
  // decomposed+nukta) depending on which tool/source produced the text.
  // Without normalizing both sides the same way, an exact string
  // comparison silently fails even though the text is "the same" to a
  // human reader -- this is what was leaving উপজেলা auto-fill blank for
  // some upazilas (কুড়িগ্রাম সদর, নাগেশ্বরী, ফুলবাড়ী and others sharing
  // that character) even when Nominatim's response was correct.
  return (s||"").toString().normalize('NFC')
    .replace(/\s*(district|zila|division|উপজেলা|জেলা|বিভাগ)\s*/gi,"")
    .replace(/ী/g,"ি").replace(/ৌ/g,"ো").replace(/ূ/g,"ু") // common vowel-sign variants across Bengali data sources
    .replace(/\s+/g,"")
    .trim().toLowerCase();
}
function _findKeyMatch(candidates,keys){
  for(var i=0;i<candidates.length;i++){
    var c=_norm(candidates[i]); if(!c)continue;
    for(var j=0;j<keys.length;j++){ if(_norm(keys[j])===c) return keys[j]; }
  }
  // fallback: substring match
  for(var i=0;i<candidates.length;i++){
    var c=_norm(candidates[i]); if(!c)continue;
    for(var j=0;j<keys.length;j++){ var k=_norm(keys[j]); if(k&&(c.indexOf(k)>=0||k.indexOf(c)>=0)) return keys[j]; }
  }
  return null;
}
window.autoSelectAdminFromGeo=function(addr){
  if(!addr) return;
  try{
    // Division (Nominatim "state" is typically "X Division" / "X বিভাগ")
    var divMatch=_findKeyMatch([addr.state],BD_DIVISIONS);
    if(divMatch){document.getElementById("fDivision").value=divMatch;}

    // District → also tells us which DAE Region contains it
    var districtCandidates=[addr.state_district,addr.district,addr.city,addr.county];
    var distMatch=_findKeyMatch(districtCandidates,Object.keys(BD_UPAZILA));
    if(distMatch){
      var regionMatch=null;
      Object.keys(BD).forEach(function(r){ if(BD[r].indexOf(distMatch)>=0) regionMatch=r; });
      if(regionMatch){
        document.getElementById("fRegion").value=regionMatch;
        onFormRegionChange(); // repopulates fDistrict for that region
        document.getElementById("fDistrict").value=distMatch;
        onFormDistrictChange(); // repopulates fUpazila for that district

        // Upazila (Nominatim "county"/"subdistrict" often holds this)
        var upaMatch=_findKeyMatch([addr.county,addr.subdistrict,addr.suburb],BD_UPAZILA[distMatch]||[]);
        if(upaMatch) document.getElementById("fUpazila").value=upaMatch;
      }
    }
  }catch(e){console.error("Auto admin-select failed:",e)}
};

function coachmarkGetVisibleNavBtn(tab){
  var btns=document.querySelectorAll('.nav-btn[data-tab="'+tab+'"]');
  for(var i=0;i<btns.length;i++){
    var r=btns[i].getBoundingClientRect();
    if(r.width>0 && r.height>0) return btns[i];
  }
  return null;
}

function coachmarkPosition(step){
  var target=coachmarkGetVisibleNavBtn(step.tab);
  if(!target) return false;
  var r=target.getBoundingClientRect();
  var pad=8;
  var ring=document.getElementById("coachmarkRing");
  ring.style.top=(r.top-pad)+"px";
  ring.style.left=(r.left-pad)+"px";
  ring.style.width=(r.width+pad*2)+"px";
  ring.style.height=(r.height+pad*2)+"px";

  var tip=document.getElementById("coachmarkTooltip");
  var tipRect=tip.getBoundingClientRect();
  var spaceBelow=window.innerHeight-r.bottom;
  var top=(spaceBelow>tipRect.height+24) ? (r.bottom+16) : Math.max(12,r.top-tipRect.height-16);
  var left=Math.min(Math.max(12,r.left+r.width/2-tipRect.width/2),window.innerWidth-tipRect.width-12);
  tip.style.top=top+"px";
  tip.style.left=left+"px";
  return true;
}

// ── Personnel directory (SAAO / monitoring officer autocomplete) ──
// Backed by User_Profile via ?directory=1 rather than a separate sheet —
// see AppsScript.gs getDirectory_() for why. Cached in localStorage with
// the same freshness pattern as national_entries_cache_v1.
var DIRECTORY_CACHE_KEY="personnel_directory_cache_v1";
var DIRECTORY_CACHE_MAX_AGE_MS=6*60*60*1000; // 6 hours — a personnel list changes rarely
var _directoryPeople=null;

function loadCachedDirectory(){
  try{
    var raw=localStorage.getItem(DIRECTORY_CACHE_KEY);
    if(!raw) return null;
    var parsed=JSON.parse(raw);
    return (parsed&&Array.isArray(parsed.people))?parsed:null;
  }catch(e){return null}
}
function saveDirectoryCache(people){
  try{localStorage.setItem(DIRECTORY_CACHE_KEY,JSON.stringify({people:people,cachedAt:new Date().toISOString()}))}catch(e){}
}
function populateDirectoryDatalists(){
  var saaoDL=document.getElementById("saaoDatalist"), officerDL=document.getElementById("officerDatalist");
  if(!saaoDL||!officerDL||!_directoryPeople) return;
  saaoDL.innerHTML=""; officerDL.innerHTML="";
  _directoryPeople.forEach(function(p){
    if(!p.name) return;
    var opt=document.createElement("option"); opt.value=p.name;
    // SAAO role -> SAAO datalist; anything else (AAO/AEO/UAO/ADD etc.) -> officer datalist
    if(p.shortRole==="SAAO") saaoDL.appendChild(opt.cloneNode(true));
    else officerDL.appendChild(opt.cloneNode(true));
  });
}
function fetchDirectory(force){
  var cached=!force?loadCachedDirectory():null;
  if(cached){
    _directoryPeople=cached.people;
    populateDirectoryDatalists();
    var ageMs=Date.now()-new Date(cached.cachedAt).getTime();
    if(ageMs<DIRECTORY_CACHE_MAX_AGE_MS) return;
  }
  fetch(GAS_SYNC_ENDPOINT+"?directory=1")
    .then(function(res){return res.json()})
    .then(function(data){
      if(!data||!data.ok||!Array.isArray(data.people)) return;
      _directoryPeople=data.people;
      saveDirectoryCache(data.people);
      populateDirectoryDatalists();
    })
    .catch(function(err){console.warn("Directory fetch failed (using cache if any):",err)});
}
