function toGASRows(base, seedlings){
  if(!seedlings || seedlings.length === 0){
    seedlings = [{ speciesName:"", category:"", quantity:0 }];
  }
  return seedlings.map(function(s){
    return Object.assign({}, base, {
      speciesName: s.speciesName || "",
      category:    s.category || "",
      quantity:    s.quantity || 0
    });
  });
}

async function sendToGAS(rows){
  try {
    var res = await fetch(GAS_SYNC_ENDPOINT, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(rows)
    });
    if(!res.ok){
      return { ok:false, error: "HTTP " + res.status };
    }
    return await res.json();
  } catch(e){
    return { ok:false, error: e.message };
  }
}

window.reverseGeocode = function(lat, lng, targetId) {
  var addrField = document.getElementById(targetId);
  if (!addrField) return;
  addrField.placeholder = "🌐 জিপিএস থেকে ঠিকানা লোড হচ্ছে...";
  
  var url = "https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=" + lat + "&lon=" + lng + "&accept-language=bn,en";
  fetch(url, {
    headers: {
      "User-Agent": "PlantationTracker/1.0"
    }
  })
  .then(function(res) { return res.json(); })
  .then(function(data) {
    if (data && data.display_name) {
      var addressStr = data.display_name;
      if (data.address) {
        var parts = [];
        var a = data.address;
        
        // Match standard local/regional divisions
        var village = a.village || a.suburb || a.neighbourhood || a.hamlet || a.road || "";
        var upazilaLoc = a.county || a.subdistrict || "";
        var districtLoc = a.state_district || a.district || a.city || "";
        var postOffice = a.postcode ? "পোস্টকোড: " + a.postcode : "";
        
        if (village) parts.push(village);
        if (upazilaLoc) parts.push(upazilaLoc);
        if (districtLoc) parts.push(districtLoc);
        if (postOffice) parts.push(postOffice);
        
        if (parts.length > 0) {
          addressStr = parts.join(", ");
        }
        if (targetId === "fAddress" && window.autoSelectAdminFromGeo) {
          window.autoSelectAdminFromGeo(a);
        }
      }
      addrField.value = addressStr;
    }
  })
  .catch(function(err) {
    console.error("Geocoding failed:", err);
  });
};

window.onGeoLocationChange = function(inputEl, targetAddressId) {

  if (m) {
    var lat = parseFloat(m[1]).toFixed(7);
    var lng = parseFloat(m[2]).toFixed(7);
    window.reverseGeocode(lat, lng, targetAddressId);
  }
};

function _setFGeoFromLatLng(lat, lng, exact, accuracy){
  // `exact` preserves whatever precision was typed/tapped rather than forcing
  // a fixed rounding, per the "bound to follow exact input" requirement.
  var latStr = exact ? String(lat) : Number(lat).toFixed(7);
  var lngStr = exact ? String(lng) : Number(lng).toFixed(7);
  var geoField = document.getElementById("fGeo");
  if(geoField) geoField.value = latStr + ", " + lngStr;
  var accEl = document.getElementById("fGeoAccuracy");
  if(accEl && typeof accuracy === "number") accEl.value = accuracy;
  var accDisplay = document.getElementById("geoAccuracyDisplay");
  if(accDisplay && typeof accuracy === "number"){accDisplay.textContent="±"+accuracy.toFixed(0)+"m";accDisplay.style.display=""}
  var latManual = document.getElementById("fGeoLatManual");
  var lngManual = document.getElementById("fGeoLngManual");
  if(latManual) latManual.value = latStr;
  if(lngManual) lngManual.value = lngStr;
}

function _ensureGeoManualMap(){
  if(geoManualMap) return geoManualMap;
  var el = document.getElementById("fGeoMap");
  if(!el || typeof L === "undefined") return null;

  var startLat = 23.8103, startLng = 90.4125, startZoom = 7;
  var existing = (_val("fGeo") || "").split(",");
  var eLat = parseFloat(existing[0]), eLng = parseFloat(existing[1]);
  if(!isNaN(eLat) && !isNaN(eLng)){ startLat = eLat; startLng = eLng; startZoom = 17; }

  geoManualMap = L.map("fGeoMap").setView([startLat, startLng], startZoom);
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    maxZoom: 19,
    attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>'
  }).addTo(geoManualMap);

  if(startZoom === 17){
    geoManualMarker = L.marker([startLat, startLng], {draggable:true}).addTo(geoManualMap);
    _bindGeoManualMarkerEvents();
  }

  geoManualMap.on("click", function(e){
    _placeGeoManualMarker(e.latlng.lat, e.latlng.lng, false);
  });

  // Geofence walking must draw on this same map instance.
  if(geoFencePoints.length){ _redrawGeoFence(); }

  return geoManualMap;
}

function _bindGeoManualMarkerEvents(){
  if(!geoManualMarker) return;
  geoManualMarker.on("dragend", function(){
    var ll = geoManualMarker.getLatLng();
    _setFGeoFromLatLng(ll.lat.toFixed(7), ll.lng.toFixed(7), true, undefined);
    window.reverseGeocode && window.reverseGeocode(ll.lat.toFixed(7), ll.lng.toFixed(7), "fAddress");
    window.autoFillNdvi && window.autoFillNdvi(ll.lat.toFixed(7), ll.lng.toFixed(7));
  });
}

function _placeGeoManualMarker(lat, lng, exact){

  if(!geoManualMarker){
    geoManualMarker = L.marker(latlng, {draggable:true}).addTo(map);
    _bindGeoManualMarkerEvents();
  } else {
    geoManualMarker.setLatLng(latlng);
  }
  map.setView(latlng, Math.max(map.getZoom(), 17));
  _setFGeoFromLatLng(lat, lng, exact, undefined);
  window.reverseGeocode && window.reverseGeocode(exact ? lat : Number(lat).toFixed(7), exact ? lng : Number(lng).toFixed(7), "fAddress");
  window.autoFillNdvi && window.autoFillNdvi(lat, lng);
}

// Haversine distance in meters — used to filter GPS noise while walking.
function _haversineMeters(lat1, lng1, lat2, lng2){
  var R = 6371000;
  var dLat = (lat2 - lat1) * Math.PI / 180;
  var dLng = (lng2 - lng1) * Math.PI / 180;
  var a = Math.sin(dLat/2)*Math.sin(dLat/2) + Math.cos(lat1*Math.PI/180)*Math.cos(lat2*Math.PI/180)*Math.sin(dLng/2)*Math.sin(dLng/2);
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

// Polygon area via the shoelace formula on a local equirectangular
// projection (metres) — accurate enough for small plantation plots.
function _polygonAreaSqm(points){
  if(points.length < 3) return 0;
  var lat0 = points[0].lat * Math.PI / 180;
  var mPerDegLat = 111320;
  var mPerDegLng = 111320 * Math.cos(lat0);
  var pts = points.map(function(p){ return { x: p.lng * mPerDegLng, y: p.lat * mPerDegLat }; });
  var area = 0;
  for(var i=0;i<pts.length;i++){
    var j = (i+1) % pts.length;
    area += pts[i].x * pts[j].y - pts[j].x * pts[i].y;
  }
  return Math.abs(area / 2);
}

function _redrawGeoFence(){
  var map = _ensureGeoManualMap();
  if(!map) return;
  if(geoFenceLayer){ map.removeLayer(geoFenceLayer); geoFenceLayer = null; }
  if(geoFencePoints.length < 2) return;
  var latlngs = geoFencePoints.map(function(p){ return [p.lat, p.lng]; });
  if(geoFenceWalking){
    geoFenceLayer = L.polyline(latlngs, { color: "#ea580c", weight: 3 }).addTo(map);
  } else {
    geoFenceLayer = L.polygon(latlngs, { color: "#15803d", weight: 3, fillOpacity: 0.15 }).addTo(map);
    map.fitBounds(geoFenceLayer.getBounds(), { maxZoom: 19, padding: [20,20] });
  }
}

function _updateGeoFenceInfo(){
  var infoEl = document.getElementById("geoFenceInfo");
  if(!infoEl) return;
  if(!geoFencePoints.length){
    infoEl.textContent = "এখনও কোনো সীমানা রেকর্ড করা হয়নি। হাঁটা শুরু করুন এবং ক্লাস্টারের চারপাশ ঘুরে আসুন।";
    return;
  }
  var areaSqm = _polygonAreaSqm(geoFencePoints);
  var areaDecimal = areaSqm / 40.4686; // 1 decimal (শতক) ≈ 40.4686 sqm — familiar local land unit
  var txt = geoFencePoints.length + "টি পয়েন্ট রেকর্ড হয়েছে।";
  if(geoFencePoints.length >= 3 && !geoFenceWalking){
    txt += " আনুমানিক আয়তন: " + areaSqm.toFixed(1) + " বর্গমিটার (≈ " + areaDecimal.toFixed(2) + " শতক)।";
  } else if(geoFenceWalking){
    txt += " হাঁটা চলছে...";
  }
  infoEl.textContent = txt;
}

window.toggleGeoFenceWalk = function(){
  var btn = document.getElementById("geoFenceWalkBtn");
  if(!geoFenceWalking){
    if(!navigator.geolocation){
      var infoEl = document.getElementById("geoFenceInfo");
      if(infoEl){ infoEl.textContent = "⚠️ আপনার ব্রাউজার লোকেশন সাপোর্ট করে না।"; }
      return;
    }
    _ensureGeoManualMap();
    geoFenceWalking = true;
    geoFencePoints = [];
    if(btn){ btn.textContent = "⏹ হাঁটা শেষ করুন"; btn.style.background = "#dc2626"; }
    geoFenceWatchId = navigator.geolocation.watchPosition(function(pos){
      var lat = pos.coords.latitude, lng = pos.coords.longitude, accuracy = pos.coords.accuracy;
      var last = geoFencePoints[geoFencePoints.length - 1];
      // Ignore GPS jitter: only record a new vertex once the walker has
      // actually moved at least ~2 metres from the last recorded point.
      if(!last || _haversineMeters(last.lat, last.lng, lat, lng) >= 2){
        geoFencePoints.push({ lat: lat, lng: lng, accuracy: accuracy });
        _redrawGeoFence();
        _updateGeoFenceInfo();
      }
    }, function(err){
      var infoEl = document.getElementById("geoFenceInfo");
      if(infoEl){ infoEl.textContent = "⚠️ লোকেশন ট্র্যাক করতে সমস্যা হয়েছে (কোড " + err.code + ")।"; }
    }, { enableHighAccuracy: true, maximumAge: 1000, timeout: 15000 });
  } else {
    geoFenceWalking = false;
    if(geoFenceWatchId !== null){ navigator.geolocation.clearWatch(geoFenceWatchId); geoFenceWatchId = null; }
    if(btn){ btn.textContent = "🚶 হাঁটা শুরু করুন"; btn.style.background = "#15803d"; }
    if(geoFencePoints.length >= 3){
      document.getElementById("fGeoFencePoints").value = JSON.stringify(geoFencePoints);
      document.getElementById("fGeoFenceAreaSqm").value = _polygonAreaSqm(geoFencePoints).toFixed(1);
    }
    _redrawGeoFence();
    _updateGeoFenceInfo();
  }
};

window.clearGeoFence = function(){
  if(geoFenceWalking){ window.toggleGeoFenceWalk(); }
  geoFencePoints = [];
  if(geoFenceLayer && geoManualMap){ geoManualMap.removeLayer(geoFenceLayer); geoFenceLayer = null; }
  document.getElementById("fGeoFencePoints").value = "";
  document.getElementById("fGeoFenceAreaSqm").value = "";
  _updateGeoFenceInfo();
};
